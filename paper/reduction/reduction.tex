\begin{scopebox}
\textbf{Scope.} This paper reports a \emph{conditional} result. It does \textbf{not} solve the Clay Millennium Problem, and it does not prove global regularity of the 3D Navier--Stokes equations. The single remaining hypothesis, \lean{PeriodicBKMIntegrability}, is a named Prop that is \textbf{not proved}. It is equivalent in difficulty to Clay Statement~(B).
\end{scopebox}
\begin{citebox}
\textbf{How to cite / What is claimed.} In Lean~4 with Mathlib, we give a machine-checked reduction of Clay Millennium Statement~(B) (periodic Navier--Stokes on $\TT$) to the single hypothesis \lean{PeriodicBKMIntegrability}. We prove this hypothesis is equivalent to global existence of smooth solutions for every smooth periodic initial field. The BKM criterion and $H^3$ control are proved unconditionally, using only the standard axioms. \lean{PeriodicBKMIntegrability} remains open. The equivalence of vorticity integrability with continuation of smooth solutions is classical (Beale--Kato--Majda, 1984~\cite{BKM1984}); the novelty here is the complete, \texttt{sorry}-free formalization. Cite as: B.~S.~Frohman, reduction paper, DOI \doi{10.5281/zenodo.23842295}; main paper, DOI \doi{10.5281/zenodo.23842555}.

\section{Introduction}
The Beale--Kato--Majda criterion~\cite{BKM1984} asserts that a smooth solution of the three-dimensional Navier--Stokes equations can cease to be smooth at a finite time $T$ only if $\int_0^T\norm{\omega(t)}_{L^\infty}\dd t=\infty$. Consequently, Statement~(B) of the Clay Millennium Problem~\cite{Fefferman2000} on the torus is equivalent to the time-integrability of the vorticity sup-norm for every smooth periodic initial field. In this paper we report a machine-checked version of this reduction in Lean~4: the criterion and the required $H^3$ control are proved with no hypothesis Props, and Statement~(B) is derived from the single named hypothesis \lean{PeriodicBKMIntegrability}, which remains open. Section~\ref{sec:stated} tabulates precisely what is stated and what is proved.

\section{\texorpdfstring{The periodic chain on $\mathbb{T}^3$}{The periodic chain on 𝕋³}}
Throughout, $\nu>0$, $u_0$ is smooth, periodic and divergence-free, and $(u,p)$ is the maximal smooth solution of
\begin{equation}\label{eq:ns}
\partial_t u+(u\cdot\nabla)u+\nabla p=\nu\Delta u,\qquad \Div u=0,\qquad u(0)=u_0\quad\text{on }\TT,
\end{equation}
with existence time $\Tstar$; $\omega=\curl u$. The definitions \lean{Clay_B_exact}, \lean{PeriodicBKMCriterion} and \lean{PeriodicBKMIntegrability} come from the dependency \file{fefferman-clay-navier-stokes}.

\begin{hyp}[\lean{PeriodicBKMIntegrability}~$\nu$; \textbf{open}]
For every maximal solution and every $T$, $t\mapsto\norm{\omega(t)}_{L^\infty}$ is integrable on $[0,T]$:
\begin{equation}\label{eq:bkmint}\int_0^T\norm{\omega(t)}_{L^\infty}\dd t<\infty.\end{equation}
\end{hyp}

\begin{theorem}[\lean{periodicBKMHsControl_three}\coined; unconditional]
For every $\nu$, \lean{PeriodicBKMHsControl}~$\nu$~$3$ holds. That is,
\[\Tstar=T<\infty\quad\text{and}\quad\int_0^T\norm{\omega}_{L^\infty}\dd t<\infty\quad\Longrightarrow\quad\sup_{t\in[0,T)}\norm{u(t)}_{H^3}<\infty.\]
\end{theorem}
\emph{Proof route (all formalized).} The $H^3$ energy identity reduces the claim to a commutator bound and a low-frequency vorticity bound.
\begin{itemize}
\item The $s=3$ commutator bound~\cite{KatoPonce1988} is
\[-2\sum_i\int\Bigl(1-\frac{\Delta}{4\pi^2}\Bigr)^{3}v_i\,\bigl((v\cdot\nabla)v\bigr)_i\dd x\le C\norm{\nabla v}_{L^\infty}\norm{v}_{H^3}^2\]
(\lean{periodicPhysicalCommutatorBound}\coined, \file{CommutatorAssembly.lean}).
\item The low-frequency bound is
\[\norm{V_R\nabla v}_{L^\infty}\le K(1+\log R)\norm{\omega}_{L^\infty}\qquad(R\ge2)\]
(\lean{periodicLowFreqVorticityBound}\coined, \file{LowFreqAssembly.lean}). It is proved through the Riesz form of Biot--Savart, a smooth dyadic decomposition with uniformly $L^1$-bounded block kernels, and a uniform $L^1$ bound for the de la Vall\'ee Poussin kernel.
\end{itemize}
Gr\"onwall's inequality with the $\log$ splitting then closes the bound.

\begin{theorem}[\lean{periodicBKMCriterion}\coined; unconditional]
For every $\nu$, \lean{PeriodicBKMCriterion}~$\nu$ holds. If $\nu>0$ and $\Tstar=T<\infty$, then
\[\int_0^T\norm{\omega(t)}_{L^\infty}\dd t=\infty.\]
\end{theorem}

\begin{theorem}[\lean{clay_B_exact_of_integrability}\coined; conditional]
\lean{PeriodicBKMIntegrability}~$\nu$ $\Rightarrow$ \lean{Clay_B_exact}~$\nu$.
\end{theorem}

\begin{remark}[Equivalence in difficulty]
The converse direction also holds. If Statement~(B) gives a global smooth solution, then by uniqueness the maximal solution is global and smooth, so $\norm{\omega}_{L^\infty}$ is continuous and integrable on every $[0,T]$. Hence \lean{PeriodicBKMIntegrability} is \emph{equivalent} to the periodic Clay Statement~(B); it is not a weaker, merely technical residue. This converse, from Statement~(B) itself, is argued informally here. What \emph{is} proved in Lean (\lean{periodicBKMIntegrability_iff_globalLifespan}, \file{IntegrabilityAttack.lean}) is that \lean{PeriodicBKMIntegrability} is equivalent to \lean{PeriodicGlobalLifespan}, i.e.\ $\Tstar=\infty$ for all smooth data. Its proof uses \lean{integrableOn_vorticity_of_lt_tstar}: $\int_0^T\norm{\omega}_{L^\infty}\dd t<\infty$ whenever $T<\Tstar$. This paper is therefore a verified reduction, not a proof of regularity.
\end{remark}
